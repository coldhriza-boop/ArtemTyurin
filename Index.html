// ===== ФИЗИКА И ОБНОВЛЕНИЕ =====
function update(deltaTime) {
if (gameState !== 'playing') return;
// Горизонтальное движение
player.vx = 0;
if (keys.left) {
player.vx = -MOVE_SPEED;
player.facingRight = false;
}
if (keys.right) {
player.vx = MOVE_SPEED;
player.facingRight = true;
}
// Прыжок (только если на земле)
if (keys.jump && player.onGround) {
player.vy = JUMP_POWER;
player.onGround = false;
}
// Применяем гравитацию
player.vy += GRAVITY * deltaTime;
// Обновляем позицию
player.x += player.vx * deltaTime;
player.y += player.vy * deltaTime;
// Проверяем столкновения с платформами
player.onGround = false;
for (let plat of platforms) {
if (checkCollision(player, plat)) {
// Определяем сторону столкновения
const overlapX = Math.min(player.x + player.width - plat.x, plat.x + plat.width - player.x);
const overlapY = Math.min(player.y + player.height - plat.y, plat.y + plat.height - player.y);
if (overlapY < overlapX) {
// Столкновение сверху или снизу
if (player.vy > 0 && player.y + player.height - player.vy * deltaTime <= plat.y + 5) {
// Приземление на платформу
player.y = plat.y - player.height;
player.vy = 0;
player.onGround = true;
} else if (player.vy < 0) {
// Удар головой о платформу
player.y = plat.y + plat.height;
player.vy = 0;
}
} else {
// Столкновение сбоку
if (player.vx > 0) {
player.x = plat.x - player.width;
} else if (player.vx < 0) {
player.x = plat.x + plat.width;
}
}
}
}
// Проверяем сбор монет
for (let coin of coins) {
if (!coin.collected && checkCircleCollision(player, coin.x, coin.y, COIN_RADIUS)) {
coin.collected = true;
coinsCollected++;
updateCoinCounter();
}
}
// Проверяем столкновение с шипами
for (let spike of spikes) {
if (checkCollision(player, spike)) {
gameOver();
return;
}
}
// Проверяем достижение финиша
if (checkCollision(player, finish)) {
if (coinsCollected === totalCoins) {
gameWon();
}
}
// Проверяем падение в пропасть
if (player.y > canvas.height + 100) {
gameOver();
return;
}
// Обновляем камеру (следует за игроком)
camera.x = player.x - canvas.width / 2 + player.width / 2;
camera.y = player.y - canvas.height / 2 + player.height / 2;
// Ограничиваем камеру
if (camera.x < 0) camera.x = 0;
if (camera.y < 0) camera.y = 0;
}
// ===== ПРОВЕРКА СТОЛКНОВЕНИЙ =====
// Столкновение двух прямоугольников (AABB)
function checkCollision(rect1, rect2) {
return rect1.x < rect2.x + rect2.width &&
rect1.x + rect1.width > rect2.x &&
rect1.y < rect2.y + rect2.height &&
rect1.y + rect1.height > rect2.y;
}
// Столкновение прямоугольника и круга
function checkCircleCollision(rect, cx, cy, radius) {
const closestX = Math.max(rect.x, Math.min(cx, rect.x + rect.width));
const closestY = Math.max(rect.y, Math.min(cy, rect.y + rect.height));
const distanceX = cx - closestX;
const distanceY = cy - closestY;
const distanceSquared = distanceX * distanceX + distanceY * distanceY;
return distanceSquared < radius * radius;
}
// ===== ОТРИСОВКА =====
function draw() {
// Очищаем canvas
ctx.clearRect(0, 0, canvas.width, canvas.height);
// Сохраняем состояние и применяем камеру
ctx.save();
ctx.translate(-camera.x, -camera.y);
// Рисуем платформы
ctx.fillStyle = '#8B4513';
for (let plat of platforms) {
ctx.fillRect(plat.x, plat.y, plat.width, plat.height);
// Рисуем траву сверху
ctx.fillStyle = '#228B22';
ctx.fillRect(plat.x, plat.y, plat.width, 8);
ctx.fillStyle = '#8B4513';
}
// Рисуем монеты
for (let coin of coins) {
if (!coin.collected) {
ctx.fillStyle = '#FFD700';
ctx.beginPath();
ctx.arc(coin.x, coin.y, COIN_RADIUS, 0, Math.PI * 2);
ctx.fill();
ctx.strokeStyle = '#FFA500';
ctx.lineWidth = 2;
ctx.stroke();
}
}
// Рисуем шипы
ctx.fillStyle = '#808080';
for (let spike of spikes) {
// Рисуем треугольные шипы
const spikeCount = Math.floor(spike.width / 20);
for (let i = 0; i < spikeCount; i++) {
ctx.beginPath();
ctx.moveTo(spike.x + i * 20, spike.y + spike.height);
ctx.lineTo(spike.x + i * 20 + 10, spike.y);
ctx.lineTo(spike.x + i * 20 + 20, spike.y + spike.height);
ctx.closePath();
ctx.fill();
}
}
// Рисуем финиш (флаг)
ctx.fillStyle = '#8B4513';
ctx.fillRect(finish.x, finish.y, 8, finish.height);
ctx.fillStyle = '#32CD32';
ctx.beginPath();
ctx.moveTo(finish.x + 8, finish.y);
ctx.lineTo(finish.x + 50, finish.y + 20);
ctx.lineTo(finish.x + 8, finish.y + 40);
ctx.closePath();
ctx.fill();
// Рисуем игрока
ctx.fillStyle = '#FF6B6B';
ctx.fillRect(player.x, player.y, player.width, player.height);
// Рисуем глаза
ctx.fillStyle = 'white';
const eyeOffsetX = player.facingRight ? 10 : 5;
ctx.beginPath();
ctx.arc(player.x + eyeOffsetX, player.y + 15, 6, 0, Math.PI * 2);
ctx.arc(player.x + eyeOffsetX + 15, player.y + 15, 6, 0, Math.PI * 2);
ctx.fill();
// Зрачки
ctx.fillStyle = 'black';
const pupilOffset = player.facingRight ? 2 : -2;
ctx.beginPath();
ctx.arc(player.x + eyeOffsetX + pupilOffset, player.y + 15, 3, 0, Math.PI * 2);
ctx.arc(player.x + eyeOffsetX + 15 + pupilOffset, player.y + 15, 3, 0, Math.PI * 2);
ctx.fill();
// Восстанавливаем состояние
ctx.restore();
}
// ===== ИГРОВОЙ ЦИКЛ =====
function gameLoop(currentTime) {
// Вычисляем delta time в секундах
const deltaTime = Math.min((currentTime - lastTime) / 1000, 0.1);
lastTime = currentTime;
update(deltaTime);
draw();
requestAnimationFrame(gameLoop);
}
